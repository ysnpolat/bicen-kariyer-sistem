# ---------------------------------------------------------------------------
# Plesk (Linux / nginx) - Ek nginx yonergeleri
# ---------------------------------------------------------------------------
# Plesk -> Alan adi -> Apache & nginx Ayarlari -> "Ek nginx yonergeleri"
# alanina asagidakileri yapistiriniz.
#
# API_SUNUCU_IP degerini API'nin calistigi makinenin adresi ile degistiriniz.
# API ayni sunucuda calisiyorsa 127.0.0.1 yaziniz.
# ---------------------------------------------------------------------------

location /api/ {
    proxy_pass         http://API_SUNUCU_IP:5177/api/;
    proxy_http_version 1.1;
    proxy_set_header   Host              $host;
    proxy_set_header   X-Real-IP         $remote_addr;
    proxy_set_header   X-Forwarded-For   $proxy_add_x_forwarded_for;
    proxy_set_header   X-Forwarded-Proto $scheme;
    proxy_read_timeout 120s;
    client_max_body_size 25m;
}

# Statik dosyalari onbellege al (index.html haric)
location ~* \.(js|css|png|jpg|jpeg|svg|woff2?)$ {
    expires 30d;
    add_header Cache-Control "public";
}

location = /index.html {
    add_header Cache-Control "no-cache";
}
